cmake_minimum_required(VERSION 3.10)
project(p10_coupled_subspace_ndt LANGUAGES CXX)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(PCL 1.10 REQUIRED COMPONENTS common filters io registration)
find_package(Eigen3 REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBUSB REQUIRED libusb-1.0)
add_executable(p10_coupled_subspace_ndt p10_coupled_subspace_ndt.cpp)
target_compile_features(p10_coupled_subspace_ndt PRIVATE cxx_std_14)
target_include_directories(p10_coupled_subspace_ndt PRIVATE ${PCL_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIRS} ${LIBUSB_INCLUDE_DIRS})
target_compile_definitions(p10_coupled_subspace_ndt PRIVATE ${PCL_DEFINITIONS})
target_link_libraries(p10_coupled_subspace_ndt PRIVATE ${PCL_LIBRARIES} ${LIBUSB_LIBRARIES})
enable_testing()
add_test(NAME p10_coupling_self_test COMMAND p10_coupled_subspace_ndt --self-test)
add_test(NAME p10_pipeline_self_test COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_pipeline.py)
set_tests_properties(p10_coupling_self_test PROPERTIES ENVIRONMENT "LD_LIBRARY_PATH=/lib/x86_64-linux-gnu")
