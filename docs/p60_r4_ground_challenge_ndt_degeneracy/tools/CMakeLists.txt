cmake_minimum_required(VERSION 3.10)
project(p60_r4_offline_screen LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 14)
find_package(PCL 1.10 REQUIRED COMPONENTS common io filters registration kdtree)
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(FRONTEND "${REPO}/src/dog_prior_map_fastlio2_frontend_exp")
include_directories(${PCL_INCLUDE_DIRS} "${FRONTEND}/include")
add_definitions(${PCL_DEFINITIONS})
add_executable(test_cloud test_cloud.cpp)
target_link_libraries(test_cloud ${PCL_LIBRARIES} usb-1.0)
enable_testing()
add_test(NAME frozen_cloud_contract COMMAND test_cloud)
add_executable(p60_r4_screen screen.cpp "${FRONTEND}/src/coupled_ndt_shadow.cpp")
target_compile_options(p60_r4_screen PRIVATE -ffunction-sections -fdata-sections)
set_target_properties(p60_r4_screen PROPERTIES LINK_FLAGS "-Wl,--gc-sections")
target_link_libraries(p60_r4_screen ${PCL_LIBRARIES} usb-1.0)
