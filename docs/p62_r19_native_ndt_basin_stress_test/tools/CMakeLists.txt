cmake_minimum_required(VERSION 3.10)
project(p62_r19_native_ndt_basin LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
find_package(PCL 1.10 REQUIRED COMPONENTS common io registration filters kdtree)
include_directories(${PCL_INCLUDE_DIRS})
add_executable(prescreen prescreen.cpp)
target_link_libraries(prescreen ${PCL_LIBRARIES} usb-1.0)
add_executable(prepare_map prepare_map.cpp)
target_link_libraries(prepare_map ${PCL_LIBRARIES} usb-1.0)
add_executable(run_basin run_basin.cpp)
target_link_libraries(run_basin ${PCL_LIBRARIES} usb-1.0)
add_executable(probe_stability probe_stability.cpp)
target_link_libraries(probe_stability ${PCL_LIBRARIES} usb-1.0)
