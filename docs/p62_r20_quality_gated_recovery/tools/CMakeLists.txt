cmake_minimum_required(VERSION 3.10)
project(p62_r20_quality_recovery LANGUAGES CXX)

find_package(PCL 1.10 REQUIRED COMPONENTS common io registration filters kdtree)

target_link_libraries(run_quality ${PCL_LIBRARIES} usb-1.0)
