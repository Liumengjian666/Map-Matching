cmake_minimum_required(VERSION 3.10)

add_executable(inspect_scans inspect_scans.cpp)
target_link_libraries(inspect_scans ${PCL_LIBRARIES} usb-1.0)

target_link_libraries(prepare_map ${PCL_LIBRARIES} usb-1.0)
