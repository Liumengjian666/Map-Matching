cmake_minimum_required(VERSION 3.10)
project(p62_r6_shadow LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 14)
find_package(PCL 1.10 REQUIRED COMPONENTS common io filters registration kdtree)
include_directories(${PCL_INCLUDE_DIRS})
add_definitions(${PCL_DEFINITIONS})
add_executable(shadow shadow.cpp)
target_link_libraries(shadow ${PCL_LIBRARIES} usb-1.0)
add_executable(test_contracts test_contracts.cpp)
target_link_libraries(test_contracts ${PCL_LIBRARIES} usb-1.0)
enable_testing()
add_test(NAME frozen_math_and_native_pcl_parity COMMAND test_contracts)
