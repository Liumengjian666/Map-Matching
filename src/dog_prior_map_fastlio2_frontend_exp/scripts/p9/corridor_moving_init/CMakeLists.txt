cmake_minimum_required(VERSION 3.10)
project(p9_corridor_moving_init LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 14)
set(FASTLIO2_ROOT "" CACHE PATH "Pinned read-only FAST-LIO2")
execute_process(COMMAND git -C "${FASTLIO2_ROOT}" rev-parse HEAD
  OUTPUT_VARIABLE PIN OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE RC)
if(NOT RC EQUAL 0 OR NOT PIN STREQUAL "7cc4175de6f8ba2edf34bab02a42195b141027e9")
  message(FATAL_ERROR "FAST-LIO2 pin mismatch")
endif()
find_package(Eigen3 REQUIRED)
find_package(OpenMP REQUIRED)
find_package(PCL 1.10 REQUIRED COMPONENTS common filters registration)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(PKG "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
add_library(moving_input_core STATIC "${PKG}/src/p7_replay_io.cpp"
  "${PKG}/src/fastlio2_frontend_ikfom.cpp" "${PKG}/src/registration_geometry.cpp"
  "${PKG}/src/scan_processor.cpp" "${PKG}/src/lidar_deskew_geometry.cpp")
target_include_directories(moving_input_core PUBLIC "${PKG}/include" PRIVATE "${FASTLIO2_ROOT}/include")
target_link_libraries(moving_input_core PUBLIC Eigen3::Eigen OpenMP::OpenMP_CXX)
enable_testing()
add_test(NAME moving_preintegration_contract COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/test_preintegration.py")
add_executable(p9_bootstrap_lidar bootstrap_lidar.cpp)
target_include_directories(p9_bootstrap_lidar PRIVATE ${PCL_INCLUDE_DIRS})
target_link_libraries(p9_bootstrap_lidar PRIVATE moving_input_core ${PCL_LIBRARIES})
foreach(TEST moving_initialization_test p7_replay_io_test scan_end_deskew_runtime_test fastlio2_propagation_contract_test registration_geometry_test)
  add_executable(${TEST} "${PKG}/tests/${TEST}.cpp")
  target_link_libraries(${TEST} PRIVATE moving_input_core)
  add_test(NAME ${TEST} COMMAND ${TEST})
endforeach()
