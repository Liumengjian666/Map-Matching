cmake_minimum_required(VERSION 3.10)
project(p9_corridor_robust_bootstrap_v2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 14)
find_package(Eigen3 REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(PCL 1.10 REQUIRED COMPONENTS common filters registration kdtree)
set(PKG "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
add_executable(p9_bootstrap_v2 bootstrap_v2.cpp "${PKG}/src/p7_replay_io.cpp")
target_include_directories(p9_bootstrap_v2 PRIVATE "${PKG}/include" ${PCL_INCLUDE_DIRS})
target_link_libraries(p9_bootstrap_v2 PRIVATE Eigen3::Eigen ${PCL_LIBRARIES})
enable_testing()
add_test(NAME robust_bootstrap_geometry_time_test COMMAND p9_bootstrap_v2 --self-test)
add_test(NAME robust_bootstrap_preparation_test COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/test_prepare.py")
add_test(NAME robust_bootstrap_motion_adapter_test COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/test_motion.py")
add_test(NAME inherited_preintegration_test COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/../corridor_moving_init/test_preintegration.py")
add_test(NAME robust_bootstrap_one_shot_test COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/test_one_shot.py")
