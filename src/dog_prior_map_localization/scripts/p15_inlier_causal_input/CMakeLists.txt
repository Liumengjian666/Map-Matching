cmake_minimum_required(VERSION 3.16)
project(p15_inlier_causal_input LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EXISTS "${INLIER_SOURCE}/cpp/inlier_core/CMakeLists.txt")
  message(FATAL_ERROR "Set INLIER_SOURCE to pinned external author source")
endif()
add_subdirectory("${INLIER_SOURCE}/cpp/inlier_core" author_core)
find_package(PCL REQUIRED COMPONENTS common io)
add_executable(inlier_causal_probe probe.cpp)
target_include_directories(inlier_causal_probe PRIVATE ${PCL_INCLUDE_DIRS})
target_link_libraries(inlier_causal_probe PRIVATE inlier_core ${PCL_LIBRARIES}
  /usr/lib/x86_64-linux-gnu/libusb-1.0.so)
target_compile_options(inlier_causal_probe PRIVATE -Wall -Wextra)
