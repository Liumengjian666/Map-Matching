cmake_minimum_required(VERSION 3.10)
project(p32_corridor02_geometry LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 14)
find_package(PCL REQUIRED COMPONENTS io registration filters kdtree)
add_executable(first_geometry first_geometry.cpp)
target_include_directories(first_geometry PRIVATE ${PCL_INCLUDE_DIRS})
target_compile_definitions(first_geometry PRIVATE ${PCL_DEFINITIONS})
find_library(SYSTEM_USB NAMES libusb-1.0.so.0 PATHS /lib/x86_64-linux-gnu NO_DEFAULT_PATH REQUIRED)
target_link_libraries(first_geometry ${PCL_LIBRARIES} ${SYSTEM_USB})
