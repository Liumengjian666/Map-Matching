cmake_minimum_required(VERSION 3.10)
project(p32_first_segment_transfer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 14)
find_package(catkin REQUIRED COMPONENTS roscpp rosbag sensor_msgs pcl_conversions)
find_package(PCL REQUIRED COMPONENTS io registration filters kdtree)
find_library(SYSTEM_USB NAMES libusb-1.0.so.0 PATHS /lib/x86_64-linux-gnu NO_DEFAULT_PATH REQUIRED)
foreach(name initializer input_bridge math_test)
  add_executable(${name} ${name}.cpp)
  target_include_directories(${name} PRIVATE ${catkin_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS})
  target_link_libraries(${name} ${catkin_LIBRARIES} ${PCL_LIBRARIES} ${SYSTEM_USB})
endforeach()
